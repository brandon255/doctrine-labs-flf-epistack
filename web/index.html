How many times was the world actually observed here?

Not how many things were cited. Those are different numbers, and they differ by more than people expect. This runs in one command on your machine, with or without a local model, and tells you the gap between the two.

Built for the Future of Life Foundation's Epistack contest.

Reviewing this? Start here — nothing to install

The engine that does the epistemic work runs with Node and nothing else: the three-level independence model, mechanical citation verification, and all eight challenger checks. A local model is optional, and only drives the proposal step and one of the two challengers.

Independence has three levels

Counting citations overstates what you know. Counting documents still overstates it, because two judges watching the same debate are not two independent witnesses.

1 — claimone excerptevery quotation counts as a source
2 — documentone PDF, one posttwo accounts of one event look independent
3 — lineageone underlying event—

The headline number is level 3, the most conservative count available. The other two stay visible underneath, because the gap between them is the interesting part.

Adjudication layers on top: a model proposes an answer with cited work, every citation is checked against the corpus mechanically, with no model involved, and a challenge panel argues against the conclusion. The deterministic challenger on that panel — eight structural checks — always runs, with or without a model installed. You decide what stands.

Run it yourself

No build step. No dependencies. Just Node 20+.

# from the repo root
npm run epistemic:covid   # → 21 excerpts, 8 documents, 3 lineages  (7x inflation)

    async function load() {
      try {
        const res = await fetch('/api/cases');
        const data = await res.json();
        const grid = document.getElementById('case-cards');
        grid.innerHTML = '';
        for (const c of data.cases) {
          const a = document.createElement('a');
          a.className = 'card';
          a.href = `/case.html?case=${encodeURIComponent(c.id)}`;
          const summary = c.summary || {};
          const count = summary.assessment_line || (c.error ? 'error' : '—');
          a.innerHTML = `
            ${c.id}
            ${escapeHtml(c.title || c.id)}
            ${summary.claim_count ?? '?'} excerpts → ${summary.document_count ?? '?'} documents → ${summary.lineage_count ?? '?'} lineages
            ${summary.inflation_factor > 1 ? `${summary.inflation_factor}x citation inflation` : ''}
          `;
          grid.appendChild(a);
        }
      } catch (e) {
        document.getElementById('case-cards').innerHTML = `
failed to load: ${escapeHtml(e.message)}
`;
      }

      try {
        const r = await fetch('/api/llm/status');
        const d = await r.json();
        const b = document.getElementById('llm-banner');
        if (d.ready) {
          b.innerHTML = `
Local LLM ready — model ${escapeHtml(d.model)}. Adjudicated assessment, chat, and alias proposals are live. Open a case to run one.
`;
        } else if (d.ok) {
          b.innerHTML = `
Ollama is up but model '${escapeHtml(d.model || '')}' is not pulled. Run: ollama pull ${escapeHtml(d.model || 'llama3.1')}
`;
        } else {
          b.innerHTML = `
No local model — that's a supported way to run this. The counts, the graphs, the reports, and all eight mechanical challenger checks need no model at all. To add the model proposal and the model challenger: Ollama + ollama pull llama3.1. To see those steps without installing anything, read docs/transcripts/.
`;
        }
      } catch (e) {
        /* silent */
      }
    }
    function escapeHtml(s){return String(s).replace(/[&<>"']/g,m=>({'&':'&','<':'<','>':'>','"':'"',"'":'''}[m]));}
    load();

    const fi = document.getElementById('file-input');
    const ta = document.getElementById('paste-input');
    const btn = document.getElementById('analyze-btn');

    });

      ta.value = `- **HIGH** The safety record holds under audit https://anthropic.com/safety
- **HIGH** The same conclusion appears in the linked report https://anthropic.com/safety
- **MEDIUM** An outside review concurs https://arxiv.org/abs/2401.00001
- A second commentary repeats the first claim https://anthropic.com/safety
- A third article quotes the second https://anthropic.com/safety`;
      lastFile = null;
      status.textContent = 'Loaded sample (5 excerpts, 3 unique hosts → expect 1.5–2x inflation).';
    });

    btn.addEventListener('click', async () => {
      status.textContent = 'Analyzing…';
      results.hidden = true;
      try {
        let res;
        if (lastFile) {
          const fd = new FormData();
          fd.append('file', lastFile);
          res = await fetch('/api/analyze', { method: 'POST', body: fd });
        } else if (ta.value.trim()) {
          res = await fetch('/api/analyze', {
            method: 'POST',
            headers: { 'content-type': 'application/json' },
            body: JSON.stringify({ text: ta.value }),
          });
        } else {
          status.textContent = 'Drop a file or paste text first.';
          return;
        }
        const payload = await res.json();
        if (!payload.ok) {
          status.innerHTML = `${escapeHtml(payload.error)}${payload.hint ? ' · ' + escapeHtml(payload.hint) : ''}`;
          return;
        }
        document.getElementById('r-block').textContent = payload.summary.block_count;
        document.getElementById('r-doc').textContent = payload.summary.document_count;
        document.getElementById('r-lin').textContent = payload.summary.lineage_count;
        document.getElementById('r-inf').textContent = payload.summary.inflation_factor + 'x';
        document.getElementById('r-line').textContent = payload.assessment_line;
        document.getElementById('r-raw').textContent = JSON.stringify(payload, null, 2);
        results.hidden = false;
        status.textContent = `Done · source: ${payload.source}${payload.filename ? ' · ' + payload.filename : ''}`;
      } catch (e) {
        status.innerHTML = `${escapeHtml(e.message)}`;
      }
    });

    async function loadAisi() {
      // Render the 9 lab cards
      const grid = document.getElementById('aisi-cards');
      try {
        const res = await fetch('/api/aisi/companies');
        const data = await res.json();
        if (!data.ok) {
          grid.innerHTML = `
failed to load AISI: ${escapeHtml(data.error)}
`;
          return;
        }
        grid.innerHTML = '';
        // Sort by inflation factor desc — most exposed first
        const sorted = [...data.companies].sort((a, b) => (b.summary?.inflation_factor ?? 0) - (a.summary?.inflation_factor ?? 0));
        for (const c of sorted) {
          const a = document.createElement('button');
          a.type = 'button';
          a.className = 'card card-button';
          a.dataset.company = c.id;
          if (c.error) {
            a.innerHTML = `${escapeHtml(c.id)}error: ${escapeHtml(c.error)}`;
          } else {
            const s = c.summary;
            const pill = s.inflation_factor >= 3
              ? `${s.inflation_factor}x inflation`
              : s.inflation_factor >= 1.5
              ? `${s.inflation_factor}x inflation`
              : `${s.inflation_factor}x inflation`;
            a.innerHTML = `
              ${escapeHtml(c.id)}
              ${escapeHtml(c.title)}
              ${s.block_count} excerpts → ${s.document_count} docs → ${s.lineage_count} lineages
              ${pill}
            `;
            a.addEventListener('click', () => loadAisiCompany(c.id));
          }
          grid.appendChild(a);
        }
      } catch (e) {
        grid.innerHTML = `
failed to load: ${escapeHtml(e.message)}
`;
      }

      // Render shared roots
      const shared = document.getElementById('aisi-shared-roots');
      try {
        const res = await fetch('/api/aisi/cross-company');
        const data = await res.json();
        if (!data.ok) {
          shared.innerHTML = `
failed to load shared roots: ${escapeHtml(data.error)}
`;
          return;
        }
        if (!data.shared_roots.length) {
          shared.innerHTML = `
No hosts cited by 2+ companies — strong independence.
`;
          return;
        }
        shared.innerHTML = ''
          + data.shared_roots.slice(0, 25).map((r) => `
            
          `).join('')
          + '
HostCompaniesTotal blocks

              ${escapeHtml(r.host)}
              ${r.company_count} (${escapeHtml(r.companies.join(', '))})
              ${r.block_count}
            
';
      } catch (e) {
        shared.innerHTML = `
${escapeHtml(e.message)}
`;
      }
    }
